<script>
(function () {
  var scale = 1, tx = 0, ty = 0;
  var dragging = false, startX, startY, startTx, startTy;

  function canvas()   { return document.getElementById('erm-modal-canvas'); }
  function viewport() { return document.getElementById('erm-modal-viewport'); }

  function applyTransform() {
    var c = canvas();
    if (!c) return;
    c.style.transform =
      'translate(calc(-50% + ' + tx + 'px), calc(-50% + ' + ty + 'px)) scale(' + scale + ')';
  }

  function zoom(factor, pivotX, pivotY) {
    var prev = scale;
    scale = Math.min(10, Math.max(0.15, scale * factor));
    var r = scale / prev;
    tx = pivotX + (tx - pivotX) * r;
    ty = pivotY + (ty - pivotY) * r;
    applyTransform();
  }

  function openModal() {
    var src = document.querySelector('#erm-diagram-wrapper svg');
    if (!src) {
      alert('Diagram not yet rendered — please wait a moment and try again.');
      return;
    }
    var c = canvas();
    c.innerHTML = '';
    var clone = src.cloneNode(true);
    clone.removeAttribute('style');
    var r = src.getBoundingClientRect();
    clone.style.width  = (r.width  || 900) + 'px';
    clone.style.height = (r.height || 600) + 'px';
    c.appendChild(clone);
    scale = 1; tx = 0; ty = 0;
    applyTransform();
    document.getElementById('erm-modal').style.display = 'block';
    document.body.style.overflow = 'hidden';
  }

  function closeModal() {
    document.getElementById('erm-modal').style.display = 'none';
    document.body.style.overflow = '';
  }

  document.addEventListener('DOMContentLoaded', function () {
    document.getElementById('erm-fullscreen-btn').addEventListener('click', openModal);
    document.getElementById('erm-modal-close').addEventListener('click', closeModal);
    document.getElementById('erm-modal-backdrop').addEventListener('click', closeModal);
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape') closeModal(); });

    document.getElementById('erm-zoom-in').addEventListener('click',    function () { zoom(1.3,  0, 0); });
    document.getElementById('erm-zoom-out').addEventListener('click',   function () { zoom(1/1.3, 0, 0); });
    document.getElementById('erm-zoom-reset').addEventListener('click', function () { scale = 1; tx = 0; ty = 0; applyTransform(); });

    var vp = viewport();

    vp.addEventListener('wheel', function (e) {
      e.preventDefault();
      var rect = vp.getBoundingClientRect();
      zoom(e.deltaY < 0 ? 1.12 : 1/1.12,
           e.clientX - rect.left  - rect.width  / 2,
           e.clientY - rect.top   - rect.height / 2);
    }, { passive: false });

    vp.addEventListener('mousedown', function (e) {
      if (e.button !== 0) return;
      dragging = true;
      startX = e.clientX; startY = e.clientY;
      startTx = tx; startTy = ty;
      vp.classList.add('dragging');
    });
    document.addEventListener('mousemove', function (e) {
      if (!dragging) return;
      tx = startTx + (e.clientX - startX);
      ty = startTy + (e.clientY - startY);
      applyTransform();
    });
    document.addEventListener('mouseup', function () {
      if (!dragging) return;
      dragging = false;
      var v = viewport();
      if (v) v.classList.remove('dragging');
    });
  });
})();
</script>
